/**
 * 验收检查脚本（开发期验证用，不进入运行环境）。
 * 用 esbuild 打包后以 node 执行：校验三个词库、单元结构、题目生成规则。
 */
import { WordBank, buildUnits } from '../src/core/wordBank';
import { QuestionGenerator } from '../src/core/questionGenerator';
import type { LevelDef, LevelsConfig, WordEntry } from '../src/types';

import { readFileSync } from 'node:fs';
const dataUrl = (f: string) => new URL(`../public/data/${f}`, import.meta.url);

const cfg = JSON.parse(readFileSync(dataUrl('levels.json'), 'utf8')) as LevelsConfig;
const failures: string[] = [];
const ok = (name: string, cond: boolean, extra = '') => {
  if (cond) console.log(`✅ ${name}${extra ? ' · ' + extra : ''}`);
  else failures.push(name);
};

const rand = (n: number) => Math.floor(Math.random() * n);

function checkLevel(level: LevelDef) {
  console.log(`\n=== ${level.name}（${level.id}）===`);
  const raw = JSON.parse(readFileSync(dataUrl(level.wordsFile), 'utf8')) as { words: WordEntry[] };
  const bank = WordBank.fromParsed(raw, level.wordCount, level.id);

  // 1. 词条数量与唯一性
  ok(`词库 ${level.wordCount} 个唯一词条`, bank.words.length === level.wordCount, `实际 ${bank.words.length}`);

  // 2. 分页
  const pageSizes = new Map<number, number>();
  for (const w of bank.words) pageSizes.set(w.globalPage, (pageSizes.get(w.globalPage) ?? 0) + 1);
  const maxPage = Math.max(...pageSizes.keys());
  ok(`${level.pageCount} 个新词页`, pageSizes.size === level.pageCount, `实际 ${pageSizes.size}`);
  const lastSize = pageSizes.get(maxPage) ?? 0;
  ok('最后一页不超过 10 词', lastSize > 0 && lastSize <= 10, `第 ${maxPage} 页 ${lastSize} 词`);
  let badPage = 0;
  for (const [p, n] of pageSizes) if (n !== 10 && p !== maxPage) badPage++;
  ok('其余每页 10 词', badPage === 0, `异常页 ${badPage}`);

  // 3. 单元结构
  const units = buildUnits(level.pageCount, 3, level.dayCount);
  const news = units.filter((u) => u.kind === 'new').length;
  const reviews = units.filter((u) => u.kind === 'review').length;
  const tests = units.filter((u) => u.kind === 'test').length;
  ok(`新词页 ${level.pageCount} / 复习页 ${reviews} / Day 测验 ${level.dayCount}`, news === level.pageCount && tests === level.dayCount, `合计 ${units.length} 个单元`);

  // 4. 每个新词页都能取到词
  let unitPageOk = true;
  for (const u of units.filter((x) => x.kind === 'new')) {
    if (bank.getPageWords(u.globalPage!).length === 0) unitPageOk = false;
  }
  ok('新词页与数据 globalPage 一致', unitPageOk);

  // 5. 全部词可搜索
  let searchOk = true;
  for (const w of bank.words) {
    if (!bank.search(w.word).some((x) => x.id === w.id)) {
      searchOk = false;
      break;
    }
  }
  ok(`全部 ${level.wordCount} 词可被搜索到`, searchOk);

  // 6. 题目生成
  const gen = new QuestionGenerator(bank);
  let questionOk = true;
  let checked = 0;
  let spellingChecked = 0;
  const sample = [...bank.words].sort(() => Math.random() - 0.5).slice(0, 600);
  for (const w of sample) {
    for (const type of gen.availableTypes(w)) {
      const q = gen.makeQuestion(w, type, new Set([w.id]), new Set());
      checked++;
      if (type === 'letter-fill') {
        const miss = q.prompt.missing ?? '';
        const run = '＿'.repeat(miss.length);
        const rebuilt = q.prompt.masked?.replace(run, miss);
        if (rebuilt !== w.word || !q.prompt.masked?.includes(run)) {
          questionOk = false;
          console.log('  letter-fill 掩码错误:', w.word, q.prompt.masked, q.prompt.missing);
        }
      }
      if (type === 'letter-fill' || type === 'letter-tiles' || type === 'spell-typing') spellingChecked++;
      if (type === 'context-cloze') {
        const s = q.prompt.sentence ?? '';
        const filled = s.replace('____', w.word).toLowerCase();
        if (!s.includes('____') || !filled.includes(w.word.toLowerCase())) {
          questionOk = false;
          console.log('  语境题挖空异常:', w.word, s);
        }
      }
      if (type === 'spell-typing') {
        if (q.answerText !== w.word || q.options !== null || q.answerKey !== null) {
          questionOk = false;
          console.log('  自由拼写题结构错误:', w.word);
        }
      }
      if (q.options) {
        const texts = q.options.map((o) => o.text.trim().toLowerCase());
        if (new Set(texts).size !== texts.length) {
          questionOk = false;
          console.log('  选项重复:', w.word, type, texts);
        }
        const correctCount = q.options.filter((o) => o.key === q.answerKey).length;
        if (correctCount !== 1 || q.options.length !== 4) {
          questionOk = false;
          console.log('  选项/答案异常:', w.word, type, correctCount, q.options.length);
        }
        if (type === 'word-to-meaning') {
          const norm = (s: string) => s.replace(/\s+/g, '').replace(/[（(].*?[)）]/g, '').replace(/[，,；;、。.!！?？]/g, '');
          const norms = q.options.map((o) => norm(o.text));
          if (new Set(norms).size !== norms.length) {
            questionOk = false;
            console.log('  中文释义重复:', w.word, norms);
          }
        }
        if (type === 'phonetic-to-word') {
          const phon = w.phonetic.trim().toLowerCase();
          for (const o of q.options) {
            if (o.key === q.answerKey) continue;
            const other = bank.words.find((x) => x.word.toLowerCase() === o.text.toLowerCase());
            if (other && other.phonetic.trim().toLowerCase() === phon) {
              questionOk = false;
              console.log('  音标题多解:', w.word, other.word);
            }
          }
        }
      } else if (type === 'letter-tiles') {
        const sorted1 = [...q.tiles!].sort().join('');
        const sorted2 = [...q.answerText!.split('')].sort().join('');
        if (sorted1 !== sorted2) {
          questionOk = false;
          console.log('  字母块与单词字母不一致:', w.word, q.tiles, q.answerText);
        }
      }
    }
  }
  ok(`题目生成检查（${checked} 题，含 ${spellingChecked} 道拼写题）`, questionOk);

  // 7. 短词不生成拼写题
  let shortOk = true;
  let longSpellOk = true;
  for (const w of bank.words) {
    const avail = gen.availableTypes(w);
    const hasSpell = avail.includes('letter-tiles') || avail.includes('spell-typing');
    if (w.word.length <= 2 && hasSpell) shortOk = false;
    if (w.word.length >= 3 && !hasSpell) longSpellOk = false;
  }
  ok('长度 ≤2 的单词无拼写题', shortOk);
  ok('长度 ≥3 的单词有拼写题型', longSpellOk);

  // 7.5 例句覆盖率（语境题依赖例句）
  const withExample = bank.words.filter((w) => Boolean(w.example)).length;
  const coverage = Math.round((withExample / bank.words.length) * 100);
  ok(`例句覆盖 ${coverage}%（≥90%）`, coverage >= 90, `${withExample} / ${bank.words.length}`);

  // 7.6 高频排序：freqRank 连续，第一页是核心高频词（没有被压到后面的基础词/虚词）
  const ranksOk = bank.words.every((w, i) => w.freqRank === i + 1 && w.sourceOrder === i + 1);
  ok('学习顺序已按高频排序（freqRank 连续）', ranksOk);
  const firstPage = bank.getPageWords(1);
  const firstOk = firstPage.length > 0 && firstPage.every((w) => (w.zipf ?? 0) > 0 && (w.zipf ?? 0) < 4.6);
  ok('第一页全部是核心高频词（无 the/a 这类基础词）', firstOk, firstPage.slice(0, 5).map((w) => w.word).join(' '));
  const zipfOk = bank.words.filter((w) => typeof w.zipf === 'number').length;
  ok(`词频字段完整（${zipfOk}/${bank.words.length}）`, zipfOk === bank.words.length);

  // 8. Day 分布：每 Day 150 词，最后一天为余数
  const daySizes = new Map<number, number>();
  for (const w of bank.words) daySizes.set(w.day, (daySizes.get(w.day) ?? 0) + 1);
  const days = [...daySizes.keys()].sort((a, b) => a - b);
  const dayOk =
    days.length === level.dayCount &&
    days.every((d) => (d === level.dayCount ? (daySizes.get(d) ?? 0) <= 150 : daySizes.get(d) === 150));
  ok(`Day 01~${String(level.dayCount - 1).padStart(2, '0')} 各 150 词，最后一天 ${daySizes.get(level.dayCount)} 词`, dayOk);

  return { bank, pageSizes };
}

ok('levels.json 有三个分级', cfg.levels.length === 3, cfg.levels.map((l) => l.name).join(' / '));

for (const level of cfg.levels) {
  checkLevel(level);
}

// 三套词库互不重复覆盖（同一个词可以在不同级别出现，但 id 前缀必须区分）
const ids = new Set<string>();
let idOk = true;
for (const level of cfg.levels) {
  const raw = JSON.parse(readFileSync(dataUrl(level.wordsFile), 'utf8')) as { words: WordEntry[] };
  for (const w of raw.words.slice(0, 50)) {
    if (ids.has(w.id)) idOk = false;
    ids.add(w.id);
  }
}
ok('三个词库的 id 互不冲突', idOk);

// 每日一诵数据
const daily = JSON.parse(readFileSync(dataUrl('daily.json'), 'utf8')) as {
  passages: { id: string; title: string; sentences: { en: string; zh: string }[]; keywords: { word: string; meaning: string }[] }[];
};
ok('每日一诵至少 10 篇', daily.passages.length >= 10, `实际 ${daily.passages.length} 篇`);
const passageOk = daily.passages.every(
  (p) =>
    p.title.trim().length > 0 &&
    p.sentences.length >= 4 &&
    p.sentences.every((s) => s.en.trim().length > 0 && s.zh.trim().length > 0) &&
    p.keywords.length > 0,
);
ok('每篇都有标题 / ≥4 句中英对照 / 关键词', passageOk);
const idsOk = new Set(daily.passages.map((p) => p.id)).size === daily.passages.length;
ok('每日一诵 id 唯一', idsOk);

console.log('\n' + (failures.length === 0 ? '🎉 全部验收项通过' : `❌ 失败 ${failures.length} 项：` + failures.join('；')));

/** 按日期从词库例句里抽 5 条（同一天结果固定，换天就换） */
function pickDailySentences(
  words: WordEntry[],
  day: number,
  count = 5,
): { en: string; zh: string; word: string; meaning: string }[] {
  const pool = words.filter((w) => w.example && w.example.length >= 25 && w.exampleTranslation);
  if (pool.length === 0) return [];
  const out: { en: string; zh: string; word: string; meaning: string }[] = [];
  const used = new Set<number>();
  let idx = (day * 131 + 17) % pool.length;
  for (let i = 0; i < pool.length && out.length < count; i++) {
    idx = (idx + 97) % pool.length;
    if (used.has(idx)) continue;
    used.add(idx);
    const w = pool[idx];
    out.push({ en: w.example, zh: w.exampleTranslation, word: w.word, meaning: w.meaning });
  }
  return out;
}

function safeHost(url: string): string {  try {
    return new URL(url).hostname.replace(/^www\./, '');
  } catch {
    return 'news';
  }

  const a = text.split('');

export function syllables(word: string): string[] {
  const w = word.toLowerCase().replace(/[^a-z]/g, '');
  if (w.length < 3) return [word];
  const vowels = 'aeiouy';
  const parts: string[] = [];
  let i = 0;
  while (i < w.length) {
    let j = i;
    while (j < w.length && !vowels.includes(w[j])) j++;
    if (j >= w.length) {
      parts.push(w.slice(i));
      break;
    }
    let k = j;
    while (k < w.length && vowels.includes(w[k])) k++;
    let m = k;
    while (m < w.length && !vowels.includes(w[m])) m++;
    if (m >= w.length) {
      parts.push(w.slice(i));
      break;
    }
    // 两个以上辅音就从中间切开，一个辅音就留给下一个音节
    const cut = m - k >= 2 ? k + 1 : k;
    parts.push(w.slice(i, cut));
    i = cut;
  }
  const out = parts.filter(Boolean);
  return out.length > 1 ? out : [word];
}

function levenshtein(a: string, b: string): number {
  const m = a.length;
  const n = b.length;
  if (m === 0) return n;
  if (n === 0) return m;
  let prev = Array.from({ length: n + 1 }, (_, j) => j);
  for (let i = 1; i <= m; i++) {
    const cur = [i];
    for (let j = 1; j <= n; j++) {
      cur[j] = Math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));
    }
    prev = cur;
  }
  return prev[n];
}

function similarity(a: string, b: string): number {
  if (!a || !b) return 0;
  const d = levenshtein(a, b);
  return Math.max(0, 1 - d / Math.max(a.length, b.length));
}

/** 音频反馈：WebAudio 合成音效（无外部资源），及 speechSynthesis 朗读 */
export interface VoiceDiag {
  supported: boolean;
  total: number;
  english: number;
  names: string[];
}

class AudioFeedback {
  private ctx: AudioContext | null = null;
  soundOn = true;
  speechOn = true;
  /** 没有本机英文语音包时，是否允许联网取在线发音 */
  onlineAudioOn = true;
  /** 系统语音明明有英文却读不出声时，用户可强制只用在线发音 */
  forceOnlineOnly = false;
  /** 发音口音：美音 / 英音 */
  accent: 'us' | 'uk' = 'us';
  /** 最近一次朗读失败的原因，给界面提示用 */
  lastSpeechError: string | null = null;
  private currentOnline: HTMLAudioElement | null = null;

  private ensureCtx(): AudioContext | null {
    if (!this.soundOn) return null;
    try {
      if (!this.ctx) {
        const AC = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
        if (!AC) return null;
        this.ctx = new AC();
      }
      if (this.ctx.state === 'suspended') void this.ctx.resume();
      return this.ctx;
    } catch {
      return null;
    }
  }

  private tone(freq: number, start: number, dur: number, type: OscillatorType = 'sine', gain = 0.12): void {
    const ctx = this.ensureCtx();
    if (!ctx) return;
    const t0 = ctx.currentTime + start;
    const osc = ctx.createOscillator();
    const g = ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, t0);
    g.gain.setValueAtTime(0.0001, t0);
    g.gain.exponentialRampToValueAtTime(gain, t0 + 0.015);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);
    osc.connect(g).connect(ctx.destination);
    osc.start(t0);
    osc.stop(t0 + dur + 0.05);
  }

  click(): void {
    this.tone(660, 0, 0.06, 'triangle', 0.08);
  }

  flip(): void {
    this.tone(520, 0, 0.05, 'triangle', 0.07);
    this.tone(700, 0.05, 0.06, 'triangle', 0.06);
  }

  correct(): void {
    this.tone(523.25, 0, 0.09, 'sine', 0.12);
    this.tone(659.25, 0.09, 0.12, 'sine', 0.12);
  }

  wrong(): void {
    this.tone(220, 0, 0.16, 'sine', 0.12);
    this.tone(196, 0.12, 0.2, 'sine', 0.1);
  }

  stamp(): void {
    this.tone(180, 0, 0.09, 'square', 0.07);
    this.tone(392, 0.05, 0.16, 'sine', 0.1);
  }

  static speechSupported(): boolean {
    return typeof window !== 'undefined' && 'speechSynthesis' in window;
  }

  /** 当前设备有没有可用的英文语音（没有的话朗读一定不会响） */
  static diagnose(): VoiceDiag {
    if (!AudioFeedback.speechSupported()) return { supported: false, total: 0, english: 0, names: [] };
    try {
      const voices = window.speechSynthesis.getVoices();
      const english = voices.filter((v) => /^en/i.test(v.lang));
      return {
        supported: true,
        total: voices.length,
        english: english.length,
        names: english.slice(0, 6).map((v) => `${v.name} (${v.lang})`),
      };
    } catch {
      return { supported: true, total: 0, english: 0, names: [] };
    }
  }

  private pickEnglishVoice(): SpeechSynthesisVoice | null {
    try {
      const voices = window.speechSynthesis.getVoices();
      if (!voices.length) return null;
      const want = this.accent === 'uk' ? /en-GB/i : /en-US/i;
      const other = this.accent === 'uk' ? /en-US/i : /en-GB/i;
      return (
        voices.find((v) => want.test(v.lang) && /google|microsoft|samantha|daniel|natural/i.test(v.name)) ??
        voices.find((v) => want.test(v.lang)) ??
        voices.find((v) => other.test(v.lang)) ??
        voices.find((v) => /^en/i.test(v.lang)) ??
        null
      );
    } catch {
      return null;
    }
  }

  /**
   * 朗读单词或句子。
   * 返回 'ok' 表示已交给系统朗读；'no-voice' 表示设备没有英文语音包；
   * 'unsupported' 表示浏览器不支持；'off' 表示用户关掉了朗读。
   * opts.onEnd 在朗读结束（或在线音频播放完）时回调，供「每日一诵」逐句连播用。
   * opts.forceOnline 强制走在线发音（用于设置页「测试在线发音」，绕开系统语音）。
   */
  speak(
    text: string,
    opts: { rate?: number; onEnd?: () => void; forceOnline?: boolean } = {},
  ): 'ok' | 'ok-online' | 'no-voice' | 'unsupported' | 'off' {
    if (!this.speechOn) return 'off';
    if (!text) return 'unsupported';
    if (opts.forceOnline || this.forceOnlineOnly) {
      if (!this.onlineAudioOn) {
        this.lastSpeechError = 'online-off';
        return 'no-voice';
      }
      return this.playOnline(text, opts.onEnd, 0);
    }
    if (!AudioFeedback.speechSupported()) {
      // 浏览器根本不支持语音合成 → 直接走在线发音
      if (this.onlineAudioOn) return this.playOnline(text, opts.onEnd, 0);
      this.lastSpeechError = 'unsupported';
      return 'unsupported';
    }
    const synth = window.speechSynthesis;

    const doSpeak = (voice: SpeechSynthesisVoice | null): 'ok' | 'ok-online' | 'no-voice' => {
      if (!voice) {
        // 本机没有英文语音包：能用在线发音就用，否则如实告诉用户
        if (this.onlineAudioOn) return this.playOnline(text, opts.onEnd, 0);
        this.lastSpeechError = 'no-english-voice';
        return 'no-voice';
      }
      try {
        const utter = new SpeechSynthesisUtterance(text);
        utter.voice = voice;
        utter.lang = voice.lang;
        utter.rate = opts.rate ?? 0.85;
        utter.pitch = 1;
        utter.volume = 1;
        let ended = false;
        // 有的手机「有英文 voice 但不出声」：2 秒内引擎没真正开口就走在线发音
        const watchdog = window.setTimeout(() => {
          if (ended) return;
          if (!synth.speaking && !synth.pending) {
            this.lastSpeechError = 'system-tts-silent';
            if (this.onlineAudioOn) this.playOnline(text, opts.onEnd, 0);
            else opts.onEnd?.();
          }
        }, 2000);
        utter.onend = () => {
          ended = true;
          window.clearTimeout(watchdog);
          opts.onEnd?.();
        };
        utter.onerror = (e) => {
          ended = true;
          window.clearTimeout(watchdog);
          this.lastSpeechError = (e as SpeechSynthesisErrorEvent).error || 'error';
          opts.onEnd?.();
        };
        // 不要在未播放时 cancel（部分安卓机 cancel 后本次 speak 会被吞掉）
        if (synth.speaking || synth.pending) synth.cancel();
        if (synth.paused) synth.resume();
        synth.speak(utter);
      } catch {
        /* 忽略 */
      }
      return 'ok';
    };

    // 语音包是异步加载的：没有就先等 voiceschanged（最多 1.2 秒）
    const voices = synth.getVoices();
    if (voices.length > 0) {
      return doSpeak(this.pickEnglishVoice());
    }

    this.lastSpeechError = 'loading-voices';
    let fired = false;
    const onVoices = () => {
      if (fired) return;
      fired = true;
      doSpeak(this.pickEnglishVoice());
    };
    try {
      synth.addEventListener?.('voiceschanged', onVoices, { once: true });
    } catch {
      /* 老浏览器没有 addEventListener */
    }
    window.setTimeout(onVoices, 1200);
    return 'ok';
  }

  /** 直接测在线发音（设置页用） */
  speakOnline(text: string): 'ok-online' | 'no-voice' | 'off' {
    const r = this.speak(text, { forceOnline: true });
    return r === 'ok-online' ? 'ok-online' : r === 'off' ? 'off' : 'no-voice';
  }

  /** 在线发音源：有道（按口音切美/英）→ 百度（备用） */
  private onlineUrl(text: string, sourceIndex: number): string {
    if (sourceIndex === 0) {
      // 有道：type=1 英音，type=2 美音
      const type = this.accent === 'uk' ? 1 : 2;
      return `https://dict.youdao.com/dictvoice?audio=${encodeURIComponent(text)}&type=${type}`;
    }
    return `https://fanyi.baidu.com/gettts?lan=en&text=${encodeURIComponent(text)}&spd=3&source=web`;
  }

  private readonly onlineSourceCount = 2;

  private failOnline(reason: string): void {
    this.lastSpeechError = `online-audio-${reason}`;
    try {
      window.dispatchEvent(new CustomEvent('dsh-speak-failed', { detail: { reason } }));
    } catch {
      /* 忽略 */
    }
  }

  /**
   * 在线发音兜底：设备没有英文语音包时，联网取词典发音。
   * 有道 → 百度 两个源依次尝试；播放被浏览器拦截时单独报 blocked。
   * 元素会挂到 DOM 上（部分手机浏览器对游离的 Audio 不出声）。
   */
  private playOnline(text: string, onEnd?: () => void, sourceIndex = 0): 'ok-online' | 'no-voice' {
    try {
      this.stopOnline();
      if (sourceIndex >= this.onlineSourceCount) {
        this.failOnline('network');
        return 'no-voice';
      }
      const url = this.onlineUrl(text, sourceIndex);
      const el = new Audio(url);
      el.preload = 'auto';
      el.volume = 1;
      el.muted = false;
      el.setAttribute('playsinline', '');
      el.style.position = 'fixed';
      el.style.left = '-9999px';
      el.style.width = '1px';
      el.style.height = '1px';
      el.setAttribute('aria-hidden', 'true');
      document.body.appendChild(el);
      this.currentOnline = el;

      const cleanup = () => {
        try {
          el.pause();
          el.remove();
        } catch {
          /* 忽略 */
        }
      };
      el.onended = () => {
        cleanup();
        onEnd?.();
      };
      el.onerror = () => {
        cleanup();
        // 当前源拿不到音频，换下一个
        this.playOnline(text, onEnd, sourceIndex + 1);
      };
      const p = el.play();
      if (p && typeof p.catch === 'function') {

      /* 忽略 */
    }

  recordAnswer(
    wordId: string,
    unitId: string,
    unitSeq: number,
    word: WordEntry,
    questionType: string,
    correct: boolean,
  ): void {
    this.ensureToday();
    const wp = this.getWordProgress(wordId);
    const now = Date.now();
    const wasDue = wp.nextReviewAt !== null && wp.nextReviewAt <= now;
    wp.appearances++;
    wp.lastAnsweredAt = now;
    wp.lastQuestionType = questionType;
    const isSpelling = questionType === 'letter-fill' || questionType === 'letter-tiles' || questionType === 'spell-typing';

    if (correct) {
      wp.correctCount++;
      wp.streakCorrect++;
      wp.lastCorrectAt = now;
      if (isSpelling) {
        wp.spellingDone = true;
        wp.spellingCorrectCount++;
      }
      // 跨单元答对记录（去重，保留插入顺序，截断到 12 个）
      if (!wp.correctInUnits.includes(unitId)) {
        wp.correctInUnits.push(unitId);
        if (wp.correctInUnits.length > 12) wp.correctInUnits.shift();
      }
      // 错词池：跨单元连续答对 2 次移出
      if (wp.inWrongPool) {
        if (wp.poolLastCorrectUnit !== null && wp.poolLastCorrectUnit !== unitId) {
          wp.poolCorrectStreak++;
        } else if (wp.poolLastCorrectUnit === null) {
          wp.poolCorrectStreak = 1;
        }
        wp.poolLastCorrectUnit = unitId;
        if (wp.poolCorrectStreak >= 2) {
          wp.inWrongPool = false;
          wp.poolCorrectStreak = 0;
          wp.poolLastCorrectUnit = null;
        }
      }
      wp.lastReviewUnitSeq = unitSeq;
      this.scheduleNext(wp, true, now);
      if (wasDue) this.data.daily.doneDueWords++;
    } else {
      wp.wrongCount++;
      wp.streakCorrect = 0;
      wp.lastWrongAt = now;
      wp.inWrongPool = true;
      wp.poolCorrectStreak = 0;
      wp.poolLastCorrectUnit = null;
      this.scheduleNext(wp, false, now);
      // 已掌握答错 → 立即回复习中；重置跨单元计数，之后“再跨单元答对 2 次”才能重新掌握
      if (wp.status === 'mastered') {
        wp.status = 'reviewing';
        wp.masteredOnce = true;
        wp.correctInUnits = [];
        return;
      }
    }
    wp.status = computeStatus(wp, word);
    if (wp.status === 'mastered') wp.masteredOnce = true;
    // 答几题落一次盘：中途退出/切后台也不至于把整页白答
    this.answeredSinceSave += 1;
    if (this.answeredSinceSave >= 5) this.save();
  }

  /** 艾宾浩斯排程：答对推进一个阶段，答错回到第一阶段 */
  private scheduleNext(wp: WordProgress, correct: boolean, now: number): void {
    wp.scheduledAt = now;
    if (correct) {
      const nextStage = Math.min(wp.reviewStage + 1, REVIEW_INTERVALS.length);
      wp.reviewStage = nextStage;
      wp.nextReviewAt = now + REVIEW_INTERVALS[nextStage - 1];
      wp.reviewCount++;
    } else {
      wp.lapses++;
      wp.reviewStage = 1;
      wp.nextReviewAt = now + REVIEW_INTERVALS[0];
    }
  }

      maxCombo: number;

    let stars = 1;
    if (accuracy >= 0.9) stars = 3;

    this.data.units[unitId] = unitResult;
    this.data.totalScore += result.score;
    this.data.lastUnitId = unitId;
    // 每日任务

export interface ImportResult {
  ok: boolean;
  imported: string[];
  words: number;
  error?: string;
}

/**
 * 导入进度。兼容三种格式：
 *  1. 2.0 全量包 { levels: { cet3, cet4, cet6 } }
 *  2. 1.x 单级包 { progress, settings }
 *  3. 裸 ProgressData { schema, words, ... }
 */
export function importAllProgress(json: string, levelIds: string[], fallbackLevelId = 'cet4'): ImportResult {
  let parsed: unknown;
  try {
    parsed = JSON.parse(json);
  } catch {
    return { ok: false, imported: [], words: 0, error: '不是合法的 JSON 文件' };
  }
  const obj = parsed as Record<string, unknown>;
  const imported: string[] = [];
  let words = 0;

  const put = (levelId: string, data: unknown): boolean => {
    const d = data as ProgressData;
    if (!d || typeof d !== 'object' || !d.words) return false;
    const upgraded = upgrade(d, levelId);
    localStorage.setItem(storageKey(levelId), JSON.stringify(upgraded));
    imported.push(levelId);
    words += Object.keys(upgraded.words).length;
    return true;
  };

  const levels = obj.levels as Record<string, unknown> | undefined;
  if (levels && typeof levels === 'object') {
    // 1) 全量包
    for (const id of levelIds) {
      if (levels[id]) put(id, levels[id]);
    }
  } else if (obj.progress && typeof obj.progress === 'object') {
    // 2) 单级包
    const target = (obj.settings as SettingsData | undefined)?.levelId;
    put(target && levelIds.includes(target) ? target : fallbackLevelId, obj.progress);
  } else if (obj.words && typeof obj.words === 'object') {
    // 3) 裸 ProgressData
    put(levelIds.includes(fallbackLevelId) ? fallbackLevelId : levelIds[0], obj);
  }

  if (imported.length === 0) {
    return { ok: false, imported, words, error: '文件里没有可识别的进度数据' };
  }
  const settings = obj.settings as Partial<SettingsData> | undefined;
  if (settings && typeof settings === 'object') {
    const merged = { ...loadSettings(), ...settings };

  }

  let data: unknown = null;
  if (obj.progress && typeof obj.progress === 'object') {
    data = obj.progress;
  } else if (obj.levels && typeof obj.levels === 'object') {
    const levels = obj.levels as Record<string, unknown>;
    data = levels[targetLevelId] ?? Object.values(levels)[0] ?? null;
  } else if (obj.words && typeof obj.words === 'object') {
    data = obj;
  }
  const d = data as ProgressData | null;
  if (!d || typeof d !== 'object' || !d.words) {
    return { ok: false, imported: [], words: 0, error: '文件里没有可识别的进度数据' };
  }
  const upgraded = upgrade(d, targetLevelId);
  localStorage.setItem(storageKey(targetLevelId), JSON.stringify(upgraded));
  return { ok: true, imported: [targetLevelId], words: Object.keys(upgraded.words).length };
}

export function upgrade(data: ProgressData, levelId: string): ProgressData {
  const out: ProgressData = {
    ...createEmptyProgress(levelId),
    ...data,
    schema: SCHEMA_VERSION,
    levelId,
  };
  if (!out.daily) out.daily = defaultDaily();
  if (typeof out.daily.doneDueWords !== 'number') out.daily.doneDueWords = 0;
  if (!out.dailyDate) out.dailyDate = todayStr();
  if (typeof out.totalScore !== 'number') out.totalScore = 0;
  if (!Array.isArray(out.checkins)) out.checkins = [];
  if (typeof out.orderVersion !== 'number') out.orderVersion = 2;
  for (const id in out.words) {
    const wp = out.words[id];
    if (typeof wp.reviewStage !== 'number') wp.reviewStage = 0;
    if (wp.nextReviewAt === undefined) wp.nextReviewAt = null;
    if (typeof wp.reviewCount !== 'number') wp.reviewCount = 0;
    if (typeof wp.lapses !== 'number') wp.lapses = 0;
    if (wp.scheduledAt === undefined) wp.scheduledAt = null;
  }
  return out;
}

    if (correctUnits >= 5 && (!needSpelling || wp.spellingDone)) return 'mastered';
    return 'reviewing';
  }
  return 'learning';

/** 随机工具：上限不含 */
const rand = (n: number) => Math.floor(Math.random() * n);
const shuffle = <T,>(arr: T[]): T[] => {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = rand(i + 1);
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
};

/** 词性主类：用于干扰项“同词性或词性接近”
 *  注意顺序：adv./adj. 要先于 v. 判断（"adv" 里含字母 v）；prep 不能和 num 混在一起 */
function posClass(pos: string): string {
  const p = pos.toLowerCase().replace(/\./g, '');
  if (p.includes('adv') || p.includes('adj')) return 'mod';
  if (p.includes('prep')) return 'prep';
  if (p.includes('conj')) return 'conj';
  if (p.includes('art')) return 'art';
  if (p.includes('num')) return 'num';
  if (p.includes('pron') || p.startsWith('n')) return 'noun';
  if (p.includes('v')) return 'verb';
  return 'other';
}

function normalizeMeaning(s: string): string {
  return s.replace(/\s+/g, '').replace(/[（(].*?[)）]/g, '').replace(/[，,；;、。.!！?？]/g, '');
}

const LETTERS = ['A', 'B', 'C', 'D'];

/**
 * 把「正确答案 + 3 个干扰项」打乱后按 A/B/C/D 顺序重新编号。
 * 之前是先编号再打乱，界面上会出现 A、D、B、C 这种顺序，看着乱。
 */
function assembleOptions(correctText: string, distractorTexts: string[]): {
  options: { key: string; text: string }[];
  answerKey: string;
} {
  const texts = shuffle([correctText, ...distractorTexts.slice(0, 3)]);
  const options = texts.map((text, i) => ({ key: LETTERS[i] ?? String.fromCharCode(65 + i), text }));
  return { options, answerKey: options.find((o) => o.text === correctText)?.key ?? 'A' };
}

/** 把例句里的目标词挖成 ____；词不在句子里就返回 null（这种词不出语境题） */
export function maskSentence(sentence: string, word: string): string | null {
  if (!sentence) return null;
  const stem = word.replace(/[^A-Za-z]/g, '');
  if (stem.length < 3) return null;
  const re = new RegExp(`\\b${stem}[a-z]{0,4}\\b`, 'i');
  if (!re.test(sentence)) return null;
  return sentence.replace(re, '____');
}

      case 'word-to-meaning':

      case 'meaning-to-word':

  /** 干扰词候选：同词性类优先，再按全库补齐 */

    const len = w.length;
    const hideCount = len >= 6 ? 2 : 1;
    const maxStart = len - hideCount - 1;
    const start = 1 + rand(Math.max(1, maxStart));
    const missing = w.slice(start, start + hideCount);
    const masked = w.slice(0, start) + '＿'.repeat(hideCount) + w.slice(start + hideCount);
    const distractorFrags = new Set<string>([missing]);
    const pool = shuffle(this.bank.words.filter((x) => x.id !== word.id));
    for (const other of pool) {
      if (distractorFrags.size >= 4) break;
      const ow = other.word;
      if (ow.length < hideCount + 1) continue;
      const s = rand(ow.length - hideCount);
      const frag = ow.slice(s, s + hideCount).toLowerCase();
      const miss = missing.toLowerCase();
      if (frag !== miss) distractorFrags.add(frag);
    }
    const alphabet = 'abcdefghijklmnopqrstuvwxyz';
    while (distractorFrags.size < 4) {
      let frag = '';
      for (let i = 0; i < hideCount; i++) frag += alphabet[rand(26)];
      distractorFrags.add(frag);
    }
    const frags = shuffle([...distractorFrags]);
    const options = frags.map((f, i) => ({ key: String.fromCharCode(65 + i), text: f }));
    return {

      type: 'letter-fill',

        missing,

      options,
      answerKey: options.find((o) => o.text === missing)?.key ?? 'A',
    };

import type { Question, QuestionType, SpellingMode, UnitDef, WordEntry } from '../types';
import { ProgressStore } from './progressStore';

/** 每次会话的拼写题单词上限（需求：一次不超过 20 个） */
const SPELLING_CAP = 20;
/** 词意题单词上限（英译中 + 中译英各一轮） */
const MEANING_CAP = 12;
/** 每局语境填空题上限 */
const CONTEXT_CAP = 10;
const REVISIT_GAP_MIN = 2;
const REVISIT_GAP_MAX = 3;

/** 艾宾浩斯到期复习虚拟单元 */
export const DUE_REVIEW_UNIT: UnitDef = { id: 'DUE', kind: 'review', label: '艾宾浩斯复习', day: 0, seq: 0 };
/** 错词复习虚拟单元 */
export const WRONG_REVIEW_UNIT: UnitDef = { id: 'W001', kind: 'review', label: '错词复习', day: 0, seq: 0 };

export function isVirtualUnit(id: string): boolean {
  return id === DUE_REVIEW_UNIT.id || id === WRONG_REVIEW_UNIT.id;
}

export interface QuizPlan {
  unit: UnitDef;
  /** 出题计划：全部为未答题状态时的问题列表（动态插入会改变 queue） */
  queue: Question[];
  /** 已答历史 */
  answered: { question: Question; correct: boolean; chosen: string | null }[];
  correct: number;
  wrong: number;
  combo: number;
  maxCombo: number;
  wrongWordIds: Set<string>;
  /** 已用翻小抄 */
  hintUsed: boolean;
  finished: boolean;
  /** 答对词统计（用于掌握数） */
  correctWordIds: Set<string>;
  /** 本局得分 */
  score: number;
  /** 最近一题的得分（用于飘字反馈） */
  lastGain: number;
}

/**
 * 构建一个学习单元的题目计划。
 *
 * 顺序固定为：① 英译中 → ② 中译英 → ③ 拼写（字母贴纸 / 自由拼写）
 * 拼写词按「今天学的词 + 艾宾浩斯到期词」挑选，每次最多 20 个。
 */
export class QuizPlanner {
  private bank: WordBank;
  private gen: QuestionGenerator;
  private scheduler: ReviewScheduler;
  private store: ProgressStore;
  private unit: UnitDef;

  constructor(bank: WordBank, store: ProgressStore, unit: UnitDef) {
    this.bank = bank;
    this.store = store;
    this.gen = new QuestionGenerator(bank);
    this.scheduler = new ReviewScheduler(bank, store);
    this.unit = unit;
  }

  build(): QuizPlan {
    const queue =
      this.unit.id === DUE_REVIEW_UNIT.id
        ? this.buildDueReview()
        : this.unit.id === WRONG_REVIEW_UNIT.id
          ? this.buildWrongReview()
          : this.unit.kind === 'new'
            ? this.buildNewPage()
            : this.unit.kind === 'review'
              ? this.buildReviewPage()
              : this.buildDayTest();
    return {
      unit: this.unit,
      queue,
      answered: [],
      correct: 0,
      wrong: 0,
      combo: 0,
      maxCombo: 0,
      wrongWordIds: new Set(),
      hintUsed: false,
      finished: false,
      correctWordIds: new Set(),
      score: 0,
      lastGain: 0,
    };
  }

  // ---------- 统一组装：英译中 → 中译英 → 语境填空 → 拼写 ----------
  private compose(meaningWords: WordEntry[], spellingWords: WordEntry[]): Question[] {
    const qs: Question[] = [];
    const used = new Set<string>();
    const usedMeanings = new Set<string>();

    // ① 英译中
    for (const w of shuffle(meaningWords)) {
      qs.push(this.gen.makeQuestion(w, 'word-to-meaning', used, usedMeanings, 'meaning-en'));
    }
    // ② 中译英
    for (const w of shuffle(meaningWords)) {
      qs.push(this.gen.makeQuestion(w, 'meaning-to-word', used, usedMeanings, 'meaning-cn'));
    }
    // ③ 语境填空（只挑有例句、且例句里能找到这个词的）
    const contextWords = shuffle(meaningWords.filter((w) => hasUsableExample(w))).slice(0, CONTEXT_CAP);
    for (const w of contextWords) {
      qs.push(this.gen.makeQuestion(w, 'context-cloze', used, usedMeanings, 'context'));
    }
    // ④ 拼写
    const spell = this.pickSpellingWords(spellingWords);
    for (const w of spell) {
      qs.push(this.gen.makeQuestion(w, this.spellingType(w), used, usedMeanings, 'spelling'));
    }
    return qs;
  }

  /** 可拼写的词：纯字母、长度 ≥3 */
  private spellable(w: WordEntry): boolean {
    return w.word.length >= 3 && /^[A-Za-z][A-Za-z'-]*$/.test(w.word);
  }

  /** 拼写题选题：先本页/本次的词，再用到期复习词补，最多 20 个 */
  private pickSpellingWords(primary: WordEntry[]): WordEntry[] {
    const out: WordEntry[] = [];
    const seen = new Set<string>();
    const take = (w: WordEntry) => {
      if (out.length >= SPELLING_CAP) return;
      if (!this.spellable(w) || seen.has(w.id)) return;
      seen.add(w.id);
      out.push(w);
    };
    for (const w of primary) take(w);
    if (out.length < SPELLING_CAP) {
      // 今天学的词优先补，再补到期复习词
      for (const w of this.scheduler.learnedTodayWords(SPELLING_CAP, seen)) take(w);
      for (const w of this.scheduler.dueWords(SPELLING_CAP, seen)) take(w);
    }
    return out;
  }

  /** 按设置决定拼写题的呈现方式 */
  private spellingType(word: WordEntry): QuestionType {
    const mode: SpellingMode = this.store.settings.spellingMode ?? 'tiles';
    if (mode === 'typing') return 'spell-typing';
    if (mode === 'tiles') return 'letter-tiles';
    // mixed：长词更容易用键盘，短词用贴纸
    if (word.word.length >= 8) return Math.random() < 0.5 ? 'spell-typing' : 'letter-tiles';
    return Math.random() < 0.5 ? 'letter-tiles' : 'spell-typing';
  }

  // ---------- 新词页 ----------
  private buildNewPage(): Question[] {
    const page = this.unit.globalPage!;
    const pageWords = this.bank.getPageWords(page).slice(0, MEANING_CAP);
    return this.compose(pageWords, pageWords);
  }

  // ---------- 复习页（按遗忘曲线到期词优先） ----------
  private buildReviewPage(): Question[] {
    const [coverFrom, coverTo] = this.unit.coversPages ?? [1, 1];
    const due = this.scheduler.dueWords(MEANING_CAP);
    const used = new Set(due.map((w) => w.id));
    const picked: WordEntry[] = [...due];
    if (picked.length < MEANING_CAP) {
      const recent = shuffle(this.scheduler.recentPageWords([coverFrom, coverTo], used));
      for (const w of recent) {
        if (picked.length >= MEANING_CAP) break;
        if (used.has(w.id)) continue;
        used.add(w.id);
        picked.push(w);
      }
    }
    // 仍不足用最弱的词补足
    if (picked.length < 6) {
      for (const w of this.scheduler.fillWrongOrWeak(MEANING_CAP - picked.length, used)) {
        if (picked.length >= MEANING_CAP) break;
        used.add(w.id);
        picked.push(w);
      }
    }
    return this.compose(picked, picked);
  }

  // ---------- 艾宾浩斯到期复习 ----------
  private buildDueReview(): Question[] {
    const due = this.scheduler.dueWords(MEANING_CAP);
    if (due.length === 0) {
      // 没有到期词时退化为最弱的词，保证页面不空
      const fallback = this.scheduler.lowestMastery(MEANING_CAP, new Set());
      const words = fallback.length > 0 ? fallback : this.bank.getPageWords(1);
      return this.compose(words.slice(0, MEANING_CAP), words);
    }
    return this.compose(due, due);
  }

  // ---------- 错词复习组 ----------
  private buildWrongReview(): Question[] {
    const wrong = this.scheduler.fillWrongOrWeak(8, new Set());
    return this.compose(wrong, wrong);
  }

  // ---------- Day 测验 ----------
  private buildDayTest(): Question[] {
    const day = this.unit.day;
    const used = new Set<string>();
    const picked: WordEntry[] = [];
    // ≥30% 历史错词
    const wrongCount = Math.ceil(MEANING_CAP * 0.3);
    for (const w of this.scheduler.fillWrongOrWeak(wrongCount, used)) {
      used.add(w.id);
      picked.push(w);
    }
    // 其余来自本 Day
    for (const w of shuffle(this.bank.getDayWords(day))) {
      if (picked.length >= MEANING_CAP) break;
      if (used.has(w.id)) continue;
      used.add(w.id);
      picked.push(w);
    }
    return this.compose(picked, picked);
  }

  /**
   * 答完一题后更新计划。答错时在间隔 2~3 题处插入该词重考题。
   * 返回更新后的 plan（同一引用）。
   */
  static recordAnswer(
    bank: WordBank,
    store: ProgressStore,
    plan: QuizPlan,
    question: Question,
    chosen: string | null,
  ): { plan: QuizPlan; correct: boolean } {
    const word = bank.getById(question.wordId)!;
    const correct =
      question.type === 'letter-tiles' || question.type === 'spell-typing'
        ? normalizeSpelling(chosen ?? '') === normalizeSpelling(question.answerText ?? '')
        : chosen === question.answerKey;
    plan.answered.push({ question, correct, chosen });
    if (plan.queue.length > 0) plan.queue.shift();
    if (correct) {
      plan.correct++;
      plan.combo++;
      plan.maxCombo = Math.max(plan.maxCombo, plan.combo);
      plan.correctWordIds.add(word.id);
      const base = 100;
      const bonus = plan.combo >= 3 ? Math.min(plan.combo * 20, 200) : 0;
      plan.lastGain = base + bonus;
      plan.score += plan.lastGain;
    } else {
      plan.wrong++;
      plan.combo = 0;
      plan.lastGain = 0;
      plan.wrongWordIds.add(word.id);
      // 插入重考：间隔 2~3 题后，换一种题型。每词每局最多 3 次
      const appearancesInSession =
        plan.answered.filter((a) => a.question.wordId === word.id).length +
        plan.queue.filter((q) => q.wordId === word.id).length;
      if (appearancesInSession < 3) {
        const gap = REVISIT_GAP_MIN + Math.floor(Math.random() * (REVISIT_GAP_MAX - REVISIT_GAP_MIN + 1));
        const insertAt = Math.min(plan.queue.length, gap);
        const gen = new QuestionGenerator(bank);
        const available = gen.availableTypes(word).filter((t) => t !== question.type);
        const retype = available.length > 0 ? available[Math.floor(Math.random() * available.length)] : question.type;
        const usedIds = new Set(plan.queue.map((q) => q.wordId));
        const revisit = gen.makeQuestion(word, retype, usedIds, new Set(), question.stage ?? 'meaning-en');
        revisit.isRevisit = true;
        plan.queue.splice(insertAt, 0, revisit);
      }
    }
    store.recordAnswer(word.id, plan.unit.id, plan.unit.seq, word, question.type, correct);
    return { plan, correct };
  }
}

function normalizeSpelling(s: string): string {
  return s.trim().toLowerCase();
}

 * 也保证「拼写方式」切换后立刻生效（不必等下一局）。
 *
 * 注意：缓存里同时存了建计划用的 bank / store 实例。切词库或重新加载后
 * 会拿到新的实例，这时必须重建，否则答题会写进已经作废的 store（记录丢失）。
 */
interface CachedPlan {
  bank: WordBank;
  store: ProgressStore;
  plan: QuizPlan;

const SESSION_CACHE = new Map<string, CachedPlan>();

export function getOrCreatePlan(bank: WordBank, store: ProgressStore, unit: UnitDef): QuizPlan {
  const key = `${store.levelId}:${unit.id}`;
  const hit = SESSION_CACHE.get(key);
  if (hit && hit.bank === bank && hit.store === store) return hit.plan;
  const plan = new QuizPlanner(bank, store, unit).build();
  SESSION_CACHE.set(key, { bank, store, plan });
  return plan;
}

export function clearPlan(levelId: string, unitId: string): void {

}

  SESSION_CACHE.clear();
}

function shuffle<T>(arr: T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

const EMPTY_PROGRESS: WordProgress = {
  id: '',
  appearances: 0,
  correctCount: 0,
  wrongCount: 0,
  lastAnsweredAt: null,
  lastQuestionType: null,

/**
 * 复习调度器：按艾宾浩斯遗忘曲线给出到期词，并负责复习页 / Day 测验 / 错词复习的选题优先级。
 */
export class ReviewScheduler {
  private bank: WordBank;
  private store: ProgressStore;

  /** 只读读取进度（不存在视为未学习） */
  private wp(id: string): WordProgress {
    return this.store.peekWordProgress(id) ?? EMPTY_PROGRESS;
  }

  constructor(bank: WordBank, store: ProgressStore) {
    this.bank = bank;
    this.store = store;
  }

  /** 到期该复习的词（按到期时间从早到晚） */
  dueWords(limit: number, excludeIds: Set<string> = new Set(), now = Date.now()): WordEntry[] {
    const out: WordEntry[] = [];
    for (const w of this.bank.words) {
      if (excludeIds.has(w.id)) continue;
      const wp = this.wp(w.id);
      if (wp.status === 'unlearned') continue;
      if (wp.nextReviewAt !== null && wp.nextReviewAt <= now) out.push(w);
    }
    out.sort((a, b) => {
      const pa = this.wp(a.id);
      const pb = this.wp(b.id);
      const na = pa.nextReviewAt ?? 0;
      const nb = pb.nextReviewAt ?? 0;
      if (na !== nb) return na - nb; // 早到期的先复习
      if (pa.lapses !== pb.lapses) return pb.lapses - pa.lapses;
      return a.sourceOrder - b.sourceOrder;
    });
    return out.slice(0, limit);
  }

  /** 到期词数量 */
  dueCount(now = Date.now()): number {
    let n = 0;
    for (const w of this.bank.words) {
      const wp = this.wp(w.id);
      if (wp.status === 'unlearned') continue;
      if (wp.nextReviewAt !== null && wp.nextReviewAt <= now) n++;
    }
    return n;
  }

  /** 今天学过的词（用于拼写题的选题池） */
  learnedTodayWords(limit: number, excludeIds: Set<string> = new Set(), now = Date.now()): WordEntry[] {
    const start = new Date(now);
    start.setHours(0, 0, 0, 0);
    const out: WordEntry[] = [];
    for (const w of this.bank.words) {
      if (excludeIds.has(w.id)) continue;
      const wp = this.wp(w.id);
      if (wp.firstLearnedAt !== null && wp.firstLearnedAt >= start.getTime()) out.push(w);
    }
    return out.slice(0, limit);
  }

  /** 复习优先级排序（从高到低） */
  rankCandidates(candidates: WordEntry[], now = Date.now()): WordEntry[] {
    return [...candidates].sort((a, b) => {
      const pa = this.wp(a.id);
      const pb = this.wp(b.id);
      // 1. 最近答错的在前
      const aWrong = pa.lastWrongAt ?? 0;
      const bWrong = pb.lastWrongAt ?? 0;
      if (aWrong !== bWrong) return bWrong - aWrong;
      // 2. 从未掌握且出错次数最多
      const aNeverMastered = pa.status !== 'mastered';
      const bNeverMastered = pb.status !== 'mastered';
      if (aNeverMastered !== bNeverMastered) return aNeverMastered ? -1 : 1;
      if (aNeverMastered && pa.wrongCount !== pb.wrongCount) return pb.wrongCount - pa.wrongCount;
      // 3. 长时间未复习的复习中词
      if (pa.status === 'reviewing' && pb.status === 'reviewing') {
        const aLast = pa.lastAnsweredAt ?? 0;
        const bLast = pb.lastAnsweredAt ?? 0;
        if (aLast !== bLast) return aLast - bLast;
      }
      // 4. 已掌握但超过 7 天未复习
      const aStale = pa.status === 'mastered' && now - (pa.lastAnsweredAt ?? 0) > 7 * 24 * 3600 * 1000;
      const bStale = pb.status === 'mastered' && now - (pb.lastAnsweredAt ?? 0) > 7 * 24 * 3600 * 1000;
      if (aStale !== bStale) return aStale ? -1 : 1;
      return a.sourceOrder - b.sourceOrder;
    });
  }

  /** 错词池中的词（按优先级） */
  wrongPoolWords(): WordEntry[] {
    const pool = this.bank.words.filter((w) => this.wp(w.id).inWrongPool);
    return this.rankCandidates(pool);
  }

  /** 历史错词（有错误记录的词） */
  historicalWrongWords(): WordEntry[] {
    const pool = this.bank.words.filter((w) => {
      const wp = this.wp(w.id);
      return wp.wrongCount > 0 || wp.inWrongPool;
    });
    return this.rankCandidates(pool);
  }

  /** 掌握度最低的词（学习过但未掌握，按答错数降序 + 答对数升序） */
  lowestMastery(limit: number, excludeIds: Set<string>): WordEntry[] {
    const pool = this.bank.words.filter((w) => {
      if (excludeIds.has(w.id)) return false;
      const wp = this.wp(w.id);
      return wp.status !== 'mastered' && wp.firstLearnedAt !== null;
    });
    return this.rankCandidates(pool).slice(0, limit);
  }

  /** 长时间未复习的“复习中”词 */
  staleReviewing(limit: number, excludeIds: Set<string>, now = Date.now()): WordEntry[] {
    const cutoff = now - 3 * 24 * 3600 * 1000;
    const pool = this.bank.words.filter((w) => {
      if (excludeIds.has(w.id)) return false;
      const wp = this.wp(w.id);
      return wp.status === 'reviewing' && (wp.lastAnsweredAt ?? 0) < cutoff;
    });
    return this.rankCandidates(pool).slice(0, limit);
  }

  /** 最近三个新词页的词 */
  recentPageWords(pages: [number, number], excludeIds: Set<string>): WordEntry[] {
    const seen = new Set<string>();
    const out: WordEntry[] = [];
    for (let p = pages[0]; p <= pages[1]; p++) {
      for (const w of this.bank.getPageWords(p)) {
        if (!excludeIds.has(w.id) && !seen.has(w.id)) {
          seen.add(w.id);
          out.push(w);
        }
      }
    }
    return out;
  }

  /** 错词不足时用掌握度最低的词补足（错词优先） */
  fillWrongOrWeak(count: number, excludeIds: Set<string>): WordEntry[] {
    const chosen: WordEntry[] = [];
    const used = new Set<string>(excludeIds);
    for (const w of this.historicalWrongWords()) {
      if (chosen.length >= count) break;
      if (used.has(w.id)) continue;
      used.add(w.id);
      chosen.push(w);
    }
    if (chosen.length < count) {
      for (const w of this.lowestMastery(count - chosen.length, used)) {
        if (chosen.length >= count) break;
        used.add(w.id);
        chosen.push(w);
      }
    }
    return chosen;
  }
}

/**
 * 按用户设定的「每页词数」重新划分词页（不改动 Day 归属，只在每个 Day 内部切页）。
 * 数据里预置的 globalPage 是按每页 10 词算的，用户可以改成 6/10/15/20。

function repaginate(words: WordEntry[], pageSize: number): WordEntry[] {
  const size = Math.max(3, Math.min(30, Math.floor(pageSize) || 10));
  const byDay = new Map<number, WordEntry[]>();
  for (const w of words) {
    const list = byDay.get(w.day);
    if (list) list.push(w);
    else byDay.set(w.day, [w]);
  }
  const out: WordEntry[] = [];
  let page = 0;
  for (const day of [...byDay.keys()].sort((a, b) => a - b)) {
    const list = byDay.get(day)!.sort((a, b) => a.sourceOrder - b.sourceOrder);
    for (let i = 0; i < list.length; i += size) {
      page += 1;
      const chunk = list.slice(i, i + size);
      const pageInDay = Math.floor(i / size) + 1;
      for (const w of chunk) out.push({ ...w, globalPage: page, pageInDay });
    }
  }
  return out;

      page.push(w);

    const words = raw.words;
    if (!Array.isArray(words)) throw new Error('词库文件结构无效：缺少 words 数组');
    if (words.length !== expectedCount) {
      throw new Error(`词条数量校验失败：实际 ${words.length}，应为 ${expectedCount}。已停止加载。`);

    const normalized = words.map(normalizeWord) as WordEntry[];
    const unique = new Set(normalized.map((w) => w.word.toLowerCase()));
    if (unique.size !== words.length) {

    return new WordBank(repaginate(normalized, pageSize), expectedCount, levelId);

/** 按词库规模计算新词页 + 复习页 + 阶段测验。
 * 学习顺序交织：每 3 个新词页后插入 1 个复习页，每个阶段末尾是测验。
 * 界面上一律用「页」说话，不出现天数。 */
export function buildUnits(totalPages: number, pagesPerReview: number, dayCount: number): UnitDef[] {
  const units: UnitDef[] = [];
  const pagesPerDay = Math.max(1, Math.ceil(totalPages / dayCount));
  let seq = 0;
  let reviewIdx = 0;
  for (let day = 1; day <= dayCount; day++) {
    const from = (day - 1) * pagesPerDay + 1;
    const to = day === dayCount ? totalPages : Math.min(day * pagesPerDay, totalPages);
    if (from > to) continue;
    const pagesInDay = to - from + 1;
    const reviewCount = Math.floor(pagesInDay / pagesPerReview);
    for (let p = from; p <= to; p++) {
      seq++;
      units.push({
        id: `N${String(p).padStart(3, '0')}`,
        kind: 'new',
        label: `第 ${p} 页`,
        day,
        seq,
        globalPage: p,
      });
      // 每满 3 个新词页（不含末尾凑不满的部分），紧跟着插一个复习页
      const pageIdx = p - from + 1;
      if (pageIdx % pagesPerReview === 0 && pageIdx / pagesPerReview <= reviewCount) {
        seq++;
        reviewIdx++;
        const coverFrom = p - pagesPerReview + 1;
        units.push({
          id: `R${String(reviewIdx).padStart(3, '0')}`,
          kind: 'review',
          label: `复习 · 第 ${coverFrom}~${p} 页`,
          day,
          seq,
          coversPages: [coverFrom, p],
        });
      }
    }
    // 阶段测验
    seq++;
    units.push({
      id: `T${String(day).padStart(2, '0')}`,
      kind: 'test',
      label: `测验 · 第 ${from}~${to} 页`,
      day,
      seq,
      coversPages: [from, to],
    });
  }
  return units;
}

export type Route =

  | { name: 'home' }
  | { name: 'directory'; day?: number }
  | { name: 'preview'; unitId: string }
  | { name: 'quiz'; unitId: string }

export function parseHash(hash: string): Route {
  const h = hash.replace(/^#\/?/, '');
  const [head, ...rest] = h.split('/');
  switch (head) {
    case '':
      return { name: 'start' };
    case 'home':
      return { name: 'home' };
    case 'directory':
      return { name: 'directory', day: rest[0] ? Number(rest[0]) : undefined };
    case 'preview':
      return { name: 'preview', unitId: rest[0] ?? '' };
    case 'quiz':
      return { name: 'quiz', unitId: rest[0] ?? '' };
    case 'result':
      return { name: 'result', unitId: rest[0] ?? '' };
    case 'wrong':
      return { name: 'wrong' };
    case 'words':
      return { name: 'words' };
    case 'word':
      return { name: 'word', wordId: rest[0] ?? '' };
    case 'daily': {
      const tab = rest[0] === 'news' ? 'news' : rest[0] === 'read' ? 'read' : 'recite';
      return { name: 'daily', tab };
    }
    case 'settings':
      return { name: 'settings' };
    default:
      return { name: 'start' };
  }
}

export function useRoute(): [Route, (r: Route) => void] {
  const [route, setRoute] = useState<Route>(() => parseHash(window.location.hash));
  useEffect(() => {
    const onHash = () => {
      setRoute(parseHash(window.location.hash));
      window.scrollTo(0, 0);
    };
    window.addEventListener('hashchange', onHash);
    return () => window.removeEventListener('hashchange', onHash);
  }, []);
  const navigate = (r: Route) => {
    const path = routeToHash(r);
    if (window.location.hash === path) {
      setRoute(r);
      window.scrollTo(0, 0);
    } else {
      window.location.hash = path;
    }
  };
  return [route, navigate];
}

export function routeToHash(r: Route): string {

    case 'start':
      return '#/';

      return '#/home';

    case 'quiz':
      return `#/quiz/${r.unitId}`;

    case 'wrong':
      return '#/wrong';
    case 'words':

      return r.tab === 'news' ? '#/daily/news' : r.tab === 'read' ? '#/daily/read' : '#/daily';

/** 题目所属阶段：先英译中 → 再中译英 → 语境填空 → 最后拼写 */
export type QuestionStage = 'meaning-en' | 'meaning-cn' | 'context' | 'spelling';

export interface Question {
  /** 会话内唯一 id */
  qid: string;
  type: QuestionType;
  stage: QuestionStage;
  wordId: string;
  /** 题干展示 */
  prompt: {
    word: string;
    phonetic: string;
    partOfSpeech: string;
    meaning: string;
    rawDefinition: string;
    /** 字母补全题的残缺展示，如 "bea_tiful" */
    masked?: string;
    /** 字母补全题缺失片段 */
    missing?: string;
    /** 语境填空题：挖空后的句子 */
    sentence?: string;
    /** 语境填空题：句子中文翻译 */
    sentenceTranslation?: string;
  };
  /** 选项（拼写题为 null） */
  options: { key: string; text: string }[] | null;
  /** 正确选项 key（拼写题为 null，正确拼写在 answerText） */
  answerKey: string | null;
  /** 字母贴纸题：打乱的字母块 */
  tiles?: string[];
  /** 拼写题：正确拼写 */
  answerText?: string;
  /** 该题是不是错词重考（间隔插入） */
  isRevisit?: boolean;

export type UnitKind = 'new' | 'review' | 'test';

export interface UnitDef {

  /** 新词页关联的 globalPage */

  maxCombo: number;

  total: number;
  masteredCount: number;
  wrongWords: string[];
  finishedAt: number;

  score: number;
  /** Day 测验 ≥90% 的金色印章 */
  goldStamp: boolean;

export interface DailyPlan {
  /** 计划新词页数 1~2 */

  /** 当天完成的单元数 */

  /** 当天接触过的词数（各单元去重后的题量） */

  /** 最近一次打卡时间戳 */
  at: number;

/** 主题色 */
export type ThemeId = 'paper' | 'mint' | 'rose' | 'night' | 'sky' | 'latte' | 'forest';

export interface ProgressData {

  levelId: string;

  units: Record<string, UnitResult>;
  /** 最近完成的单元 id */
  lastUnitId: string | null;
  /** 累计学习词数（进入过预览的词） */

  /** 连续学习天数 */

  /** 今日计划 */
  daily: DailyPlan;
  /** 计划日期 */
  dailyDate: string;
  /** 累计总得分（游戏进度） */

  checkins: CheckinDay[];
  /** 词库排序版本（2 = 高频优先） */
  orderVersion?: number;
  createdAt: number;
  updatedAt: number;

export type SpellingMode = 'tiles' | 'typing' | 'mixed';

export interface SettingsData {

  accent?: 'us' | 'uk';
  /** 主题色 */
  theme: ThemeId;
